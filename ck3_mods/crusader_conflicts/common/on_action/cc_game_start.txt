on_game_start = {
	effect = {
		every_player = {
			debug_log = "CW_SAVE_PURGE_debug_log_on_game_start"
		}
	}
	on_actions = {
		cwalert_setup_on_actions
	}
}

# Like on_game_start, except it is called once the host (or player, in single player) exits the lobby. Good for anything where you need to know who the players are, or what the game rules are
on_game_start_after_lobby = {
	on_actions = {
		cwalert_setup_after_game_start_on_actions
	}
}

cwalert_setup_after_game_start_on_actions = {
	effect = {
		every_player = { # Intro event
		 	trigger_event = cwalert.1
			debug_log = "CW_SAVE_PURGE_debug_log_on_action3"
			# CC option settings persist as character variables (cc_op_*) set by the cc_set_*
			# scripted guis in 02_cc_options.txt when the player clicks an option button in
			# cc_options.gui. The GUI button states read these variables directly, so settings
			# survive save/load with no script-side restoration needed here.
			# NOTE: ExecuteConsoleCommand is a GUI-only data function and cannot be run from
			# on_action effects (the former 'battle_max:purge' line was a parse error).
		}
	}
}
