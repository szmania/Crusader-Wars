
cc_can_assault_siege_with_troops = {
	# Check if the attacker army exists and has enough troops compared to the garrison
	# Assuming attacker_army and siege_var (for garrison) are set in the datacontext
	# A simple ratio of 3:1 attacker to garrison is used as a placeholder
	exists = scope:attacker_army
	exists = scope:siege_var
	scope:attacker_army = {
		army_size >= { value = scope:siege_var.garrison_size multiply = 3 }
	}
}
