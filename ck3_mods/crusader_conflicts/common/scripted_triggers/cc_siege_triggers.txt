
# CC-231: Disable Siege Button when insufficient troops
#
# NOTE: The gate for the siege assault button is implemented directly in
# window_siege.gui using engine-native Siege-scope accessors, NOT via a
# scripted trigger. The previous scripted-trigger approach
# (cc_can_assault_siege_with_troops) was removed because it relied on
# non-existent accessors (army_size, garrison_size) and on
# SiegeWindow.GetSiege.MakeScope, which does not exist on the Siege type.
#
# The correct gate (see window_siege.gui, button_tertiary) is:
#   enabled = "[And(Siege.EnableAssaultButton, Not(Siege.IsBlockedByGarrison))]"
#   tooltip = "[SelectLocalization( Not(Siege.IsBlockedByGarrison), Siege.GetAssaultButtonTooltip, 'cc_siege_assault_insufficient_troops_tooltip')]"
#
# Vanilla rule: a siege only starts when the attacking army's soldiers exceed
# the holding garrison. The relevant Siege-scope GUI accessors are:
#   Siege.GetAttackerMen (int32) - attacking soldiers
#   Siege.GetDefenderMen (int32) - defending garrison
#   Siege.IsBlockedByGarrison (bool) - true when the attacker lacks enough men
#   Siege.EnableAssaultButton (bool) - whether an assault is currently allowed
#
# This file is intentionally left as documentation only. No scripted trigger
# is wired to the GUI for this mechanic.
