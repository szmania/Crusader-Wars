### CC Options persistence ###
# These scripted guis store the player's CC option settings as character variables.
# Character variables persist across save/load cycles (pattern proven in 03_cc_battle_control.txt).
# cc_game_start.txt reads these variables on game start and re-applies the settings.

# Battle Scale (mutually exclusive)
cc_set_battle_scale_25 = {
	scope = character
	effect = {
		remove_variable = cc_op_battle_scale_50
		remove_variable = cc_op_battle_scale_100
		set_variable = cc_op_battle_scale_25
	}
}
